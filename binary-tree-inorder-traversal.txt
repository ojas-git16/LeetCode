vector<int> inorderTraversal(TreeNode* root) {
        vector<int> ans;
        if (root == NULL)
            return ans;
        stack<TreeNode*> st;
        TreeNode* ptr = root;
        while (ptr || !st.empty()) {
            while (ptr) {
                st.push(ptr);
public:
class Solution {
                ptr = ptr->left;
            }
            ptr = st.top();
            st.pop();
            ans.push_back(ptr->val);
            ptr = ptr->right;
        }
        return ans;
    }
};
