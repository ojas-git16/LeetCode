class Solution {
public:
    bool canJump(vector<int>& nums) {
        int n=nums.size();
        int lim = 0;
        for(int i=0; i<n && i<=lim; i++)
            lim = max(i+nums[i],lim);
    }
        if(lim>=n-1)  return true;
        return false;
};
